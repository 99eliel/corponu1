rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    function logado() {
      return request.auth != null;
    }

    function usuarioExiste() {
      return logado()
        && exists(/databases/$(database)/documents/usuarios/$(request.auth.uid));
    }

    function meuPerfil() {
      return get(/databases/$(database)/documents/usuarios/$(request.auth.uid));
    }

    function ativo() {
      return usuarioExiste()
        && meuPerfil().data.ativo == true;
    }

    function admin() {
      return ativo()
        && meuPerfil().data.tipo == "admin";
    }

    function usuarioComum() {
      // Todo perfil autenticado e ativo pode executar as rotinas operacionais.
      // O tipo "admin" continua necessário somente nas áreas administrativas.
      return ativo();
    }


    function financeiroPodeDefinirValor() {
      return admin() || (
        ativo()
        && meuPerfil().data.keys().hasAny(["permissoes"])
        && meuPerfil().data.permissoes.keys().hasAny(["recursos"])
        && (
          (
            meuPerfil().data.permissoes.recursos.keys().hasAny(["gerenciarValores"])
            && meuPerfil().data.permissoes.recursos.gerenciarValores == true
          )
          || (
            meuPerfil().data.permissoes.recursos.keys().hasAny(["marcarPagamentos"])
            && meuPerfil().data.permissoes.recursos.marcarPagamentos == true
          )
        )
      );
    }

    function alteracaoValorTotalManualPermitida() {
      return financeiroPodeDefinirValor()
        && resource.data.statusPagamento != "pago"
        && request.resource.data.statusPagamento == "pendente"
        && request.resource.data.atualizadoPor == request.auth.uid
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly([
          "total",
          "subtotal",
          "valorUnitario",
          "statusPagamento",
          "valorPendente",
          "valorManualFinanceiroPendente",
          "valorManualFinanceiro",
          "valorTotalDefinidoManualmente",
          "valorTotalManual",
          "formaValorPagamento",
          "motivoValorPendente",
          "avisoPagamento",
          "valorInformadoPor",
          "valorInformadoEm",
          "atualizadoPor",
          "atualizadoEm",
          "versaoValorManual",
          "observacoes"
        ]);
    }

    function pagamentoDoUsuario() {
      return usuarioComum()
        && resource.data.criadoPor == request.auth.uid;
    }

    function novoPagamentoDoUsuario() {
      return usuarioComum()
        && request.resource.data.criadoPor == request.auth.uid
        && request.resource.data.atualizadoPor == request.auth.uid;
    }

    function novoLogDoUsuario() {
      return usuarioComum() && (
        (
          request.resource.data.keys().hasAny(["usuarioUid"])
          && request.resource.data.usuarioUid == request.auth.uid
        )
        || (
          request.resource.data.keys().hasAny(["usuarioId"])
          && request.resource.data.usuarioId == request.auth.uid
        )
        || (
          request.resource.data.keys().hasAny(["criadoPor"])
          && request.resource.data.criadoPor == request.auth.uid
        )
      );
    }

    match /usuarios/{userId} {
      allow read: if admin() || (logado() && request.auth.uid == userId);
      allow create: if admin();
      allow update: if admin();
      allow delete: if admin();
    }

    match /produtos/{produtoId} {
      allow read: if usuarioComum();
      allow create, update, delete: if admin();
    }

    match /ordensProducao/{ordemId} {
      allow read: if usuarioComum();
      allow create, update: if usuarioComum();
      allow delete: if admin();
    }

    match /faccoes/{faccaoId} {
      allow read: if usuarioComum();
      allow create, update, delete: if admin();
    }

    match /manejos/{manejoId} {
      allow read: if usuarioComum();
      allow create, update: if usuarioComum();
      allow delete: if admin();
    }

    match /fasesManejo/{faseId} {
      allow read: if usuarioComum();
      allow create, update: if usuarioComum();
      allow delete: if admin();
    }

    match /logsAlteracoes/{logId} {
      allow read: if admin();

      // As rotinas atuais usam usuarioUid, usuarioId ou criadoPor,
      // dependendo da tela. Todos precisam apontar para o usuário logado.
      allow create: if novoLogDoUsuario();

      allow update, delete: if false;
    }

    match /configuracoes/{docId} {
      allow read: if usuarioComum();
      allow write: if admin();
    }

    match /celulas/{celulaId} {
      allow read: if usuarioComum();
      allow create, update, delete: if admin();
    }

    match /movimentacoesProducao/{movimentoId} {
      allow read: if usuarioComum();
      allow create, update: if usuarioComum();
      allow delete: if admin();
    }

    match /precosReferencia/{precoId} {
      allow read: if usuarioComum();
      allow create, update, delete: if admin();
    }

    match /entregasPagamento/{entregaId} {
      // A chegada precisa pesquisar pagamentos ligados à movimentação para
      // impedir duplicidade, inclusive quando o lançamento foi criado por
      // outro usuário. Por isso, todo usuário ativo pode consultar a coleção.
      allow read: if usuarioComum();

      // Usuário operacional cria somente pagamentos vinculados ao próprio UID.
      allow create: if admin() || novoPagamentoDoUsuario();

      // Financeiro autorizado pode informar o valor total de Sutiã Montagem/
      // Sutiã Completo. A regra limita exatamente os campos financeiros alterados.
      // Usuário operacional continua podendo corrigir apenas o próprio pagamento.
      allow update: if admin()
        || alteracaoValorTotalManualPermitida()
        || (
          pagamentoDoUsuario()
          && resource.data.statusPagamento != "pago"
          && novoPagamentoDoUsuario()
          && request.resource.data.statusPagamento != "pago"
        );

      // Exclusão pelo usuário é permitida somente para o próprio lançamento
      // pendente/sem valor. Pagamentos quitados continuam restritos ao ADM.
      allow delete: if admin() || (
        pagamentoDoUsuario()
        && resource.data.statusPagamento != "pago"
      );
    }

    match /ajustesMigracao/{ajusteId} {
      // Administrador acessa todos os ajustes. Usuário comum cria e lê
      // somente o histórico das próprias correções de local.
      allow read: if admin() || (
        usuarioComum()
        && resource.data.criadoPor == request.auth.uid
      );
      allow create: if admin() || (
        usuarioComum()
        && request.resource.data.criadoPor == request.auth.uid
      );
      allow update, delete: if admin();
    }

    match /relatoriosMigracaoLigia/{relatorioId} {
      allow read: if usuarioComum();
      allow create, update, delete: if admin();
    }

    match /datasIncoerentesLigia/{dataId} {
      allow read: if usuarioComum();
      allow create, update, delete: if admin();
    }

    match /processosMigracao/{processoId} {
      allow read: if usuarioComum();
      allow create, update, delete: if admin();
    }


    match /travasOperacionais/{travaId} {
      // Trava curta usada para impedir que duas abas/usuários confirmem
      // o mesmo envio, chegada ou pagamento ao mesmo tempo.
      allow read: if usuarioComum();

      allow create: if usuarioComum()
        && request.resource.data.criadoPor == request.auth.uid;

      // Uma trava expirada pode ser assumida por outro usuário ativo.
      // A lógica do sistema só faz isso quando o prazo já terminou.
      allow update: if usuarioComum()
        && request.resource.data.criadoPor == request.auth.uid;

      allow delete: if admin() || (
        usuarioComum()
        && resource.data.criadoPor == request.auth.uid
      );
    }
  }
}
