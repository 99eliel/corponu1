rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    function logado() {
      return request.auth != null;
    }

    function usuarioExiste() {
      return logado()
        && exists(/databases/$(database)/documents/usuarios/$(request.auth.uid));
    }

    function meuPerfil() {
      return get(/databases/$(database)/documents/usuarios/$(request.auth.uid));
    }

    function ativo() {
      return usuarioExiste()
        && meuPerfil().data.ativo == true;
    }

    function admin() {
      return ativo()
        && meuPerfil().data.tipo == "admin";
    }

    function usuarioComum() {
      return ativo()
        && meuPerfil().data.tipo in ["admin", "usuario"];
    }

    function pagamentoDoUsuario() {
      return usuarioComum()
        && resource.data.criadoPor == request.auth.uid;
    }

    function novoPagamentoDoUsuario() {
      return usuarioComum()
        && request.resource.data.criadoPor == request.auth.uid
        && request.resource.data.atualizadoPor == request.auth.uid;
    }

    match /usuarios/{userId} {
      allow read: if admin() || (logado() && request.auth.uid == userId);
      allow create: if admin();
      allow update: if admin();
      allow delete: if admin();
    }

    match /produtos/{produtoId} {
      allow read: if usuarioComum();
      allow create, update, delete: if admin();
    }

    match /ordensProducao/{ordemId} {
      allow read: if usuarioComum();
      allow create, update: if usuarioComum();
      allow delete: if admin();
    }

    match /faccoes/{faccaoId} {
      allow read: if usuarioComum();
      allow create, update, delete: if admin();
    }

    match /manejos/{manejoId} {
      allow read: if usuarioComum();
      allow create, update: if usuarioComum();
      allow delete: if admin();
    }

    match /fasesManejo/{faseId} {
      allow read: if usuarioComum();
      allow create, update: if usuarioComum();
      allow delete: if admin();
    }

    match /logsAlteracoes/{logId} {
      allow read: if admin();
      allow create: if usuarioComum()
        && request.resource.data.usuarioUid == request.auth.uid;
      allow update, delete: if false;
    }

    match /configuracoes/{docId} {
      allow read: if usuarioComum();
      allow write: if admin();
    }

    match /celulas/{celulaId} {
      allow read: if usuarioComum();
      allow create, update, delete: if admin();
    }

    match /movimentacoesProducao/{movimentoId} {
      allow read: if usuarioComum();
      allow create, update: if usuarioComum();
      allow delete: if admin();
    }

    match /precosReferencia/{precoId} {
      allow read: if usuarioComum();
      allow create, update, delete: if admin();
    }

    match /entregasPagamento/{entregaId} {
      // O administrador continua vendo e alterando todos os pagamentos.
      // O usuário comum vê somente pagamentos criados pelo próprio UID.
      allow read: if admin() || pagamentoDoUsuario();

      // Corrige o problema em que a chegada era salva, mas o pagamento
      // falhava para usuário comum. O usuário cria apenas em seu próprio nome.
      allow create: if admin() || novoPagamentoDoUsuario();

      // O usuário pode corrigir somente o próprio pagamento enquanto não pago.
      allow update: if admin() || (
        pagamentoDoUsuario()
        && resource.data.statusPagamento != "pago"
        && novoPagamentoDoUsuario()
        && request.resource.data.statusPagamento != "pago"
      );

      // Exclusão pelo usuário é permitida apenas para o próprio pagamento
      // pendente/sem valor. Pagamento já pago exige administrador.
      allow delete: if admin() || (
        pagamentoDoUsuario()
        && resource.data.statusPagamento != "pago"
      );
    }

    match /ajustesMigracao/{ajusteId} {
      // Administrador acessa todos os ajustes. Usuário comum cria e lê
      // somente o histórico das próprias correções de local.
      allow read: if admin() || (
        usuarioComum()
        && resource.data.criadoPor == request.auth.uid
      );
      allow create: if admin() || (
        usuarioComum()
        && request.resource.data.criadoPor == request.auth.uid
      );
      allow update, delete: if admin();
    }

    match /relatoriosMigracaoLigia/{relatorioId} {
      allow read: if usuarioComum();
      allow create, update, delete: if admin();
    }

    match /datasIncoerentesLigia/{dataId} {
      allow read: if usuarioComum();
      allow create, update, delete: if admin();
    }

    match /processosMigracao/{processoId} {
      allow read: if usuarioComum();
      allow create, update, delete: if admin();
    }


    match /travasOperacionais/{travaId} {
      // Trava curta usada para impedir que duas abas/usuários confirmem
      // o mesmo envio, chegada ou pagamento ao mesmo tempo.
      allow read: if usuarioComum();

      allow create: if usuarioComum()
        && request.resource.data.criadoPor == request.auth.uid;

      // Uma trava expirada pode ser assumida por outro usuário ativo.
      // A lógica do sistema só faz isso quando o prazo já terminou.
      allow update: if usuarioComum()
        && request.resource.data.criadoPor == request.auth.uid;

      allow delete: if admin() || (
        usuarioComum()
        && resource.data.criadoPor == request.auth.uid
      );
    }
  }
}
